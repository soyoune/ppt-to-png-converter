import os import comtypes.client from pptx import Presentation def pptx_to_png_with_custom_filename(pptx_path, output_dir): # 1. python-pptx로 슬라이드 밖의 텍스트 분석 prs = Presentation(pptx_path) # 파워포인트 기본 슬라이드 크기 (픽셀 또는 인치 단위 기준, 일반적으로 가로: 9144000, 세로: 5143500 등 EMU 단위 사용) slide_width = prs.slide_width slide_height = prs.slide_height file_names = [] for i, slide in enumerate(prs.slides): custom_name = f"slide_{i+1}" # 기본 파일명 (텍스트가 없을 경우) # 슬라이드 내의 모든 도형/텍스트 상자 탐색 for shape in slide.shapes: if shape.has_text_frame: # 위치 정보 확인 (EMU 단위) left = shape.left top = shape.top # 조건: 슬라이드 영역 밖(왼쪽 < 0, 오른쪽 > slide_width, 위쪽 < 0, 아래쪽 > slide_height 등)에 있는 텍스트 상자 # 여기서는 슬라이드 가로 폭(slide_width)보다 오른쪽에 있거나, 세로 폭(slide_height)보다 아래에 있는 텍스트를 파일명으로 타겟팅 if left > slide_width or top > slide_height or left < 0 or top < 0: text = shape.text_frame.text.strip() if text: # 파일명으로 쓰기 부적절한 특수문자 제거 safe_text = "".join(c for c in text if c.isalnum() or c in (' ', '_', '-')).strip() if safe_text: custom_name = safe_text break # 첫 번째로 발견된 바깥 텍스트를 파일명으로 사용 file_names.append(custom_name) # 2. PowerPoint COM 객체를 이용해 고화질 슬라이드 이미지 내보내기 (Windows 전용) os.makedirs(output_dir, exist_ok=True) abs_pptx_path = os.path.abspath(pptx_path) abs_output_dir = os.path.abspath(output_dir) powerpoint = comtypes.client.CreateObject("Powerpoint.Application") powerpoint.Visible = 1 # 파일 열기 (읽기 전용) presentation = powerpoint.Presentations.Open(abs_pptx_path, ReadOnly=1) # 각 슬라이드를 이미지로 내보내기 for i, slide in enumerate(presentation.Slides): target_name = file_names[i] output_file_path = os.path.join(abs_output_dir, f"{target_name}.png") # 슬라이드를 PNG로 내보내기 slide.Export(output_file_path, "PNG") print(f"변환 완료: {target_name}.png") presentation.Close() powerpoint.Quit() print("모든 슬라이드 변환이 완료되었습니다!") # --- 실행 예시 --- if __name__ == "__main__": pptx_file = "파워포인트_파일경로.pptx" # 변환할 PPTX 파일 경로 output_folder = "./output_images" # 이미지가 저장될 폴더 경로 pptx_to_png_custom_name(pptx_file, output_folder)
